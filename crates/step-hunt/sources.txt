# step-hunt's sources: what a campaign reads (see src/sources.rs).
#
# source <name> <root> <glob> <licence> [whole]
#
# Adding a source of third parties' files needs its licence cleared by
# the owner first (design 277, "Decisions for the owner").

# Committed: NIST's MBE PMI models (AP203, public domain) and the files
# crates/ryustep/tests/data/occt/gen.cpp made with OCCT.
source committed crates/ryustep/tests/corpus *.stp LicenseRef-public-domain whole
source occt crates/ryustep/tests/data/occt *.stp LicenseRef-ryucad whole

# Pinned, fetched by scripts/fetch-corpus.sh: NIST's whole package and
# OCCT 7.8.1's two samples (LGPL-2.1 with OCCT's exception).
source nist ${RYUCAD_CORPUS:-target/corpus}/NIST-PMI-STEP-Files **/*.stp LicenseRef-public-domain whole
source occt-pinned ${RYUCAD_CORPUS:-target/corpus}/occt *.step LGPL-2.1-only whole

# Pinned, fetched by scripts/fetch-kicad.sh: KiCad 9.0.9's 3D models
# (CC-BY-SA-4.0 with KiCad's exception).
source kicad-models ${RYUEDA_KICAD_CORPUS:-target/kicad}/kicad-packages3D-9.0.9 *.3dshapes/*.step CC-BY-SA-4.0

# Made by scripts/export-pcba.sh (in the devcontainer, after
# scripts/fetch-kicad.sh): ryudfm's KiCad boards as STEP, with their
# components' models and alone. Each board is its authors' (Apache-2.0 or
# CC-BY-4.0: its SOURCE file says which), its models KiCad's (CC-BY-SA-4.0
# with KiCad's exception).
source pcba ${HUNT_PCBA:-target/pcba} */*.step LicenseRef-pcba whole

# The owner's own STEP files (SolidWorks exports of vendor parts and the
# owner's assemblies), a `cox files` set pulled at run time into
# ${HUNT_COX_FILES:-target/cox-files}: every version up to the one named,
# deduplicated by SHA-256, each file checked against what cox lists, and
# the version's files pinned by known.txt's `file` lines. Their licences
# are unknown: nothing of them is ever committed, and issues name cases
# only by case ID, file SHA-256 and set version (design 277, 12.C). Never
# run in CI.
source owner-step cox-files:step-files@v2 **/* LicenseRef-owner-private whole
